<template>
  <div>
    <h1>Dynamic Component</h1>
    <component :is="CurrentComponent" />
    <button @click="handleSwitch('HelloWorld')">Show HelloComponent</button>
    <button @click="handleSwitch('Introduction')">Show Introduction Component</button>
  </div>
  
</template>
<script>
import { ref } from 'vue';
import HelloWorld from './components/HelloWorld.vue'
import Introduction from './components/Introduction.vue'
export default {
  components: {
    HelloWorld,
    Introduction
  },

  setup(){
    const CurrentComponent = ref('');
    const handleSwitch = (component) => {
      CurrentComponent.value = component;      
    };

    return {
      CurrentComponent,
      handleSwitch
    };
  }

}
</script>
