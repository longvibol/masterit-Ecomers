+ Dynamic Class : it need true or false (must put binding :class) : binding and class is the html attribute 

 <div :class="{ dynamic: isDynamic }">Home Using Dynamic Class</div>


where we control the true or false 

export default {
  setup() {
    const isDynamic = ref(false);

    const changeStyle = () => {
      isDynamic.value = !isDynamic.value;
    };

    return {
      isDynamic,
      changeStyle,
    };
  },
};

<template>
  <div class="container">
    <div :class="{ dynamic: isDynamic }">Home Using Dynamic Class</div>
    <div class="staic">Contact</div>
    <div class="staic">About</div>
    <button @click="changeStyle">Apply Style to dynamic class</button>
  </div>
</template>

<script>
import { ref } from "vue";

export default {
  setup() {
    const isDynamic = ref(false);

    const changeStyle = () => {
      isDynamic.value = !isDynamic.value;
    };

    return {
      isDynamic,
      changeStyle,
    };
  },
};
</script>
